Before: one running (max, sum) per lane, [XBLOCK, R0_BLOCK] live across the loop

    acc_max = full([XBLOCK, R0_BLOCK], -inf)
    acc_sum = zeros([XBLOCK, R0_BLOCK])
    for r0 in range(0, rnumel, R0_BLOCK):
        x = load(...)                                           # [XBLOCK, R0_BLOCK]
        acc_max, acc_sum = online_softmax_combine(acc_max, acc_sum, x)   # lane-wise
    row_max, row_sum = online_softmax_reduce(acc_max, acc_sum, dim=1)    # cross-lane, once

After: one running (max, sum) per row, [XBLOCK] live across the loop

    acc_max = full([XBLOCK], -inf)
    acc_sum = zeros([XBLOCK])
    for r0 in range(0, rnumel, R0_BLOCK):
        x = load(...)                                           # [XBLOCK, R0_BLOCK]
        blk_max, blk_sum = online_softmax_reduce(x, dim=1)      # cross-lane, per block
        acc_max, acc_sum = online_softmax_combine_with_sum(acc_max, acc_sum, blk_max, blk_sum)
    row_max, row_sum = acc_max, acc_sum
